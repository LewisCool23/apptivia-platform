```js
// ── Push Queue Engine ─────────────────────────────────────────

/**
 * Enqueue a push operation. Does NOT execute immediately.
 * Returns the queue item ID.
 */
async function enqueuePush(sb, {
  integrationId,
  organizationId,
  entityType,
  entityId,
  action,
  payload,
  triggeredBy = 'api',
  sourceEvent = null,
  scheduledAt = null,
}) {
  const { data, error } = await sb.from('integration_push_queue').insert({
    integration_id: integrationId,
    organization_id: organizationId,
    entity_type: entityType,
    entity_id: entityId,
    action,
    payload,
    triggered_by: triggeredBy,
    source_event: sourceEvent,
    scheduled_at: scheduledAt || new Date().toISOString(),
    status: 'pending',
  }).select('id').single();

  if (error) throw error;
  return data.id;
}

/**
 * Process pending items in the push queue.
 * Called by CronManager on a schedule (every 15 minutes recommended).
 */
async function processPushQueue(sb) {
  const BATCH_SIZE = 50;

  // Fetch pending items (or failed items with retries remaining)
  const { data: items } = await sb
    .from('integration_push_queue')
    .select('*, integrations!inner(integration_type, credentials, status, organization_id)')
    .or('status.eq.pending,and(status.eq.failed,retry_count.lt.max_retries)')
    .lte('scheduled_at', new Date().toISOString())
    .order('scheduled_at', { ascending: true })
    .limit(BATCH_SIZE);

  if (!items?.length) return { processed: 0, sent: 0, failed: 0 };

  let sent = 0, failed = 0;

  for (const item of items) {
    // Mark as processing
    await sb.from('integration_push_queue')
      .update({ status: 'processing', processing_at: new Date().toISOString() })
      .eq('id', item.id);

    try {
      // Check integration is still connected
      if (item.integrations.status !== 'connected') {
        await sb.from('integration_push_queue')
          .update({ status: 'skipped', error: 'Integration disconnected' })
          .eq('id', item.id);
        continue;
      }

      // Get provider and ensure fresh token
      const provider = getProvider(item.integrations.integration_type);
      if (!provider?.push?.[item.action]) {
        throw new Error(`Provider ${item.integrations.integration_type} has no push.${item.action} method`);
      }

      const freshIntegration = await ensureFreshToken(sb, {
        ...item.integrations,
        id: item.integration_id,
      });

      // Execute the push
      const result = await withRetry(
        () => provider.push[item.action](freshIntegration, item.payload),
        `push:${item.action}:${item.id}`,
        1  // Single attempt here; retries handled by queue
      );

      // Success
      await sb.from('integration_push_queue').update({
        status: 'sent',
        external_id: result?.externalId || null,
        response_data: result || {},
        sent_at: new Date().toISOString(),
        updated_at: new Date().toISOString(),
      }).eq('id', item.id);

      // Audit log
      await sb.from('integration_push_log').insert({
        push_queue_id: item.id,
        integration_id: item.integration_id,
        organization_id: item.organization_id,
        action: item.action,
        entity_type: item.entity_type,
        status: 'sent',
        external_id: result?.externalId || null,
        payload_summary: JSON.stringify(item.payload).slice(0, 500),
      });

      // Update entity map if external ID returned
      if (result?.externalId && item.entity_id) {
        await sb.from('integration_entity_map').upsert({
          integration_id: item.integration_id,
          organization_id: item.organization_id,
          apptivia_entity: item.entity_type,
          apptivia_id: item.entity_id,
          external_entity: item.action.replace('create', '').replace('log', '').replace('update', ''),
          external_id: result.externalId,
          synced_at: new Date().toISOString(),
        }, { onConflict: 'integration_id,apptivia_entity,apptivia_id' });
      }

      sent++;
    } catch (err) {
      const retryCount = (item.retry_count || 0) + 1;
      const isFinalFailure = retryCount >= (item.max_retries || 3);
      const backoffMinutes = Math.pow(2, retryCount) * 5; // 10min, 20min, 40min

      await sb.from('integration_push_queue').update({
        status: isFinalFailure ? 'failed' : 'failed',
        retry_count: retryCount,
        error: err.message,
        scheduled_at: isFinalFailure
          ? item.scheduled_at  // Keep original
          : new Date(Date.now() + backoffMinutes * 60000).toISOString(),
        updated_at: new Date().toISOString(),
      }).eq('id', item.id);

      // Audit log on final failure
      if (isFinalFailure) {
        await sb.from('integration_push_log').insert({
          push_queue_id: item.id,
          integration_id: item.integration_id,
          organization_id: item.organization_id,
          action: item.action,
          entity_type: item.entity_type,
          status: 'failed',
          error: err.message,
          payload_summary: JSON.stringify(item.payload).slice(0, 500),
        });
      }

      failed++;
    }
  }

  return { processed: items.length, sent, failed };
}
```

### Phase 3B: Schedule Calculation Helper (same session as 3A)

**File: `public_html/backend/server.js`** (or could go in a new `sequenceService.js`)

```js
/**
 * Calculate the next step execution time, respecting:
 * - delay_days from previous step
 * - send window (start/end time)
 * - timezone
 * - skip weekends
 */
function calculateNextStepTime(fromDate, delayDays, windowStart, windowEnd, timezone, skipWeekends) {
  // Add delay days
  const target = new Date(fromDate);
  target.setDate(target.getDate() + delayDays);
  
  // Parse window times (HH:MM format)
  const [startH, startM] = (windowStart || '09:00').split(':').map(Number);
  const [endH, endM] = (windowEnd || '17:00').split(':').map(Number);
  
  // Set to start of send window if before it
  // (Simplified: using UTC offset. Production should use a timezone library)
  const hour = target.getUTCHours();
  if (hour < startH) {
    target.setUTCHours(startH, startM, 0, 0);
  } else if (hour >= endH) {
    // Push to next day's window start
    target.setDate(target.getDate() + 1);
    target.setUTCHours(startH, startM, 0, 0);
  }
  
  // Skip weekends if configured
  if (skipWeekends) {
    const day = target.getDay();
    if (day === 0) target.setDate(target.getDate() + 1); // Sunday -> Monday
    if (day === 6) target.setDate(target.getDate() + 2); // Saturday -> Monday
  }
  
  return target;
}
```

**Note:** For production-quality timezone handling, consider adding `luxon` or using `Intl.DateTimeFormat` with timezone support. The simplified version above handles 90% of cases.

```js
/**
 * Cron: sequence-execution
 * Runs every hour. Finds active enrollments with next_step_at <= NOW(),
 * evaluates step conditions, executes the step, advances the enrollment.
 */
async function runSequenceExecution() {
  const sb = getSupabaseAdmin();
  if (!sb) return { processed: 0, sent: 0, skipped: 0, failed: 0 };
  
  try {
    const now = new Date().toISOString();
    
    // 1. Find due enrollments
    const { data: dueEnrollments } = await sb
      .from('engage_sequence_enrollments')
      .select(`
        *,
        engage_sequences!inner(
          id, name, organization_id, status,
          default_channel, send_window_start, send_window_end,
          send_timezone, skip_weekends, total_steps
        )
      `)
      .eq('status', 'active')
      .lte('next_step_at', now)
      .limit(100);
    
    if (!dueEnrollments?.length) return { processed: 0, sent: 0, skipped: 0, failed: 0 };
    
    let sent = 0, skipped = 0, failed = 0;
    
    for (const enrollment of dueEnrollments) {
      const sequence = enrollment.engage_sequences;
      
      // Skip if sequence is no longer active
      if (sequence.status !== 'active') {
        skipped++;
        continue;
      }
      
      // 2. Get current step
      const { data: step } = await sb
        .from('engage_sequence_steps')
        .select('*')
        .eq('sequence_id', enrollment.sequence_id)
        .eq('step_number', enrollment.current_step)
        .single();
      
      if (!step) {
        // No more steps -- mark enrollment as completed
        await sb.from('engage_sequence_enrollments').update({
          status: 'completed',
          completed_at: now,
        }).eq('id', enrollment.id);
        
        // Increment total_completed on sequence
        await sb.from('engage_sequences').update({
          total_completed: (sequence.total_completed || 0) + 1,
        }).eq('id', sequence.id);
        
        skipped++;
        continue;
      }
      
      // 3. Evaluate step conditions
      if (step.skip_if_replied) {
        // Check if prospect has replied to any previous step
        const { data: replies } = await sb
          .from('engage_sequence_executions')
          .select('id')
          .eq('enrollment_id', enrollment.id)
          .eq('status', 'replied')
          .limit(1);
        
        if (replies?.length) {
          // Mark enrollment as replied
          await sb.from('engage_sequence_enrollments').update({
            status: 'replied',
            completed_at: now,
          }).eq('id', enrollment.id);
          
          await sb.from('engage_sequences').update({
            total_replied: (sequence.total_replied || 0) + 1,
          }).eq('id', sequence.id);
          
          skipped++;
          continue;
        }
      }
      
      // 4. Route to channel
      try {
        let executionStatus = 'sent';
        let executionResult = {};
        
        const channel = step.channel || sequence.default_channel || 'email';
        
        if (channel === 'email') {
          // Direct email via SMTP
          if (enrollment.prospect_email) {
            const { sendEmail } = require('./emailService');
            await sendEmail({
              recipients: [enrollment.prospect_email],
              subject: step.subject || `${sequence.name} - Step ${step.step_number}`,
              html: step.body || '',
              text: step.body?.replace(/<[^>]*>/g, '') || '',
            });
          } else {
            executionStatus = 'failed';
            executionResult = { error: 'No prospect email' };
          }
        } else if (channel === 'outreach' || channel === 'salesloft') {
          // Route through CRM push queue (Gap 1 dependency)
          const providerType = channel === 'outreach' ? 'outreach' : 'salesloft';
          const { data: crmIntegration } = await sb.from('integrations')
            .select('id')
            .eq('organization_id', sequence.organization_id)
            .eq('integration_type', providerType)
            .eq('status', 'connected')
            .maybeSingle();
          
          if (crmIntegration) {
            await integrations.enqueuePush(sb, {
              integrationId: crmIntegration.id,
              organizationId: sequence.organization_id,
              entityType: 'sequence_step',
              entityId: step.id,
              action: 'createTask',
              payload: {
                title: step.subject || `${sequence.name} Step ${step.step_number}`,
                description: step.body,
                type: 'to-do',
                dueDate: now,
              },
              triggeredBy: 'sequence',
              sourceEvent: `sequence_step_${enrollment.sequence_id}_${step.step_number}`,
            });
          } else {
            executionStatus = 'failed';
            executionResult = { error: `No connected ${providerType} integration` };
          }
        } else if (channel === 'call' || channel === 'task') {
          // Create a manual task/reminder -- no automated send
          executionStatus = 'pending';
          // Could create a notification for the rep to make the call
        }
        
        // 5. Log execution
        await sb.from('engage_sequence_executions').insert({
          enrollment_id: enrollment.id,
          step_id: step.id,
          step_number: step.step_number,
          channel,
          subject: step.subject,
          body: step.body,
          status: executionStatus,
          sent_at: executionStatus === 'sent' ? now : null,
        });
        
        // 6. Advance enrollment to next step
        const nextStepNumber = enrollment.current_step + 1;
        const { data: nextStep } = await sb
          .from('engage_sequence_steps')
          .select('delay_days')
          .eq('sequence_id', enrollment.sequence_id)
          .eq('step_number', nextStepNumber)
          .maybeSingle();
        
        if (nextStep) {
          const nextStepAt = calculateNextStepTime(
            new Date(),
            nextStep.delay_days || 1,
            sequence.send_window_start,
            sequence.send_window_end,
            sequence.send_timezone,
            sequence.skip_weekends !== false
          );
          
          await sb.from('engage_sequence_enrollments').update({
            current_step: nextStepNumber,
            last_step_at: now,
            next_step_at: nextStepAt.toISOString(),
          }).eq('id', enrollment.id);
        } else {
          // No more steps -- mark completed
          await sb.from('engage_sequence_enrollments').update({
            status: 'completed',
            last_step_at: now,
            completed_at: now,
          }).eq('id', enrollment.id);
          
          await sb.from('engage_sequences').update({
            total_completed: (sequence.total_completed || 0) + 1,
          }).eq('id', sequence.id);
        }
        
        if (executionStatus === 'sent') sent++;
        else failed++;
        
      } catch (stepErr) {
        console.error(`[cron:sequence-execution] Step error for enrollment ${enrollment.id}:`, stepErr.message);
        
        // Log failed execution
        await sb.from('engage_sequence_executions').insert({
          enrollment_id: enrollment.id,
          step_id: step.id,
          step_number: step.step_number,
          channel: step.channel || 'email',
          status: 'failed',
          metadata: { error: stepErr.message },
        });
        
        failed++;
      }
    }
    
    return { processed: dueEnrollments.length, sent, skipped, failed };
  } catch (err) {
    console.error('[cron:sequence-execution] Error:', err.message);
    return { error: err.message };
  }
}
```
